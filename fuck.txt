⏺ GitHub Actions Pipeline Strategy Summary

  I've created a comprehensive, scalable GitHub Actions pipeline strategy for the Flavor mono-repo that allows you to manage multiple artifacts efficiently:

  📁 Structure Created

  1. .github/workflow-config.yml - Central configuration file that controls all workflows (can be toggled on/off)
  2. .github/workflows/orchestrator.yml - Main orchestrator workflow that reads config and runs appropriate jobs
  3. .github/workflows/release.yml - Comprehensive release pipeline for multi-platform artifacts
  4. .github/actions/setup-flavor/ - Reusable action for Python/uv setup
  5. .github/actions/build-helpers/ - Reusable action for building Go/Rust helpers

  🎯 Key Features

  Scalability

  - Configuration-driven: Single YAML file controls all workflow behavior
  - Matrix builds: Parallel builds across OS/language/architecture combinations
  - Change detection: Only runs relevant workflows based on file changes
  - Reusable actions: DRY principle with composite actions

  Artifact Management

  - Multi-platform builds: Linux, macOS (Intel & ARM), Windows
  - Multiple languages: Go and Rust helpers built in parallel
  - Python wheel: Built once and reused across platforms
  - Automatic versioning: Embeds version info in all binaries

  Release Process

  - One-click releases: Manual trigger or tag-based automation

       309 +                      delay *= 2; // Exponential backoff
       310 +                  }
       311 +                  Err(e) => return Err(e),
       312 +              }
       313 +          }
       314 +          
       315 +          unreachable!()
       316 +      }
       317 +  }

       403 +  func executeSpawnMode(workenv string, args []string, logger hclog.Logger) {
       404 +      // SPAWN MODE: Full resilience
       405 +      
       406 +      cmd := exec.Command(pythonPath, args...)
       407 +      cmd.Env = append(os.Environ(), fmt.Sprintf("FLAVOR_WORKENV=%s", workenv))
       408 +      cmd.Dir = userCwd
       409 +      cmd.Stdin = os.Stdin
       410 +      cmd.Stdout = os.Stdout
       411 +      cmd.Stderr = os.Stderr
       412 +      
       413 +      // 5. START WITH MONITORING
       414 +      if err := cmd.Start(); err != nil {
       415 +          logger.Error("Failed to start process", "error", err)
       416 +          os.Exit(1)
       417 +      }
       418 +      
       419 +      // 6. SIGNAL FORWARDING (spawn only)
       420 +      sigChan := make(chan os.Signal, 1)
       421 +      signal.Notify(sigChan, syscall.SIGTERM, syscall.SIGINT)
       422 +      
       423 +      go func() {
       424 +          sig := <-sigChan
       425 +          logger.Info("Forwarding signal to child", "signal", sig)
       426 +          cmd.Process.Signal(sig)
       427 +          
       428 +          // Wait for graceful shutdown
       429 +          time.Sleep(5 * time.Second)
       430 +          
       431 +          // Force kill if needed
       432 +          if cmd.ProcessState == nil {
       433 +              cmd.Process.Kill()
       434 +          }
       435 +      }()
       436 +      
       437 +      // 7. RESOURCE MONITORING (spawn only)
       438 +      go func() {
       439 +          ticker := time.NewTicker(10 * time.Second)
       440 +          defer ticker.Stop()
       441 +          
       442 +          for range ticker.C {
       443 +              if err := monitor.CheckResources(); err != nil {
       444 +                  logger.Warn("Resource warning", "error", err)
       445 +              }
       446 +          }
       447 +      }()
       448 +      
       449 +      // 8. WAIT AND CLEANUP
       450 +      err := cmd.Wait()
       451 +      cleanupOnExit(workenv, cmd.ProcessState.ExitCode())
       452 +      
       453 +      if err != nil {
       454 +          if exitErr, ok := err.(*exec.ExitError); ok {
       455 +              os.Exit(exitErr.ExitCode())
       456 +          }
       457 +          os.Exit(1)
       458 +      }

       461 +  func executeExecMode(workenv string, args []string, logger hclog.Logger) {
       462 +      // EXEC MODE: Limited resilience
       463 +      // Note: On Unix, this replaces the process
       464 +      // On Windows, it spawns and exits
       465 +      
       466 +      logger.Info("Executing in exec mode - limited resilience")
       467 +      
       468 +      pythonPath := filepath.Join(workenv, "venv/bin/python")
       469 +      
       470 +      // On Unix: actually replace the process
       471 +      if runtime.GOOS != "windows" {
       472 +          // Prepare environment
       473 +          env := append(os.Environ(), fmt.Sprintf("FLAVOR_WORKENV=%s", workenv))
       474 +          
       475 +          // This never returns on success
       476 +          err := syscall.Exec(pythonPath, append([]string{pythonPath}, args...), env)
       477 +          logger.Error("Exec failed", "error", err)
       478 +          os.Exit(1)
       479 +      }
       480 +      
       481 +      // On Windows: simulate exec with spawn + exit
       482 +      cmd := exec.Command(pythonPath, args...)
       483 +      cmd.Env = append(os.Environ(), fmt.Sprintf("FLAVOR_WORKENV=%s", workenv))
       484 +      cmd.Stdin = os.Stdin
       485 +      cmd.Stdout = os.Stdout
       486 +      cmd.Stderr = os.Stderr
       487 +      
       488 +      err := cmd.Run()

       491 +              os.Exit(exitErr.ExitCode())
       492 +          }
       493 +          os.Exit(1)
       494 +      }
       495 +      os.Exit(0)

       498 +  // Retry logic for extraction
       499 +  func extractWithRetry(bundlePath string, logger hclog.Logger) (string, error) {
       500 +      var lastErr error
       501 +      
       502 +      for attempt := 1; attempt <= 3; attempt++ {
       503 +          workenv, err := extractBundle(bundlePath)
       504 +          if err == nil {
       505 +              return workenv, nil
       506 +          }
       507 +          
       508 +          lastErr = err
       509 +          logger.Warn("Extraction attempt failed", "attempt", attempt, "error", err)
       510 +          
       511 +          if attempt < 3 {
       512 +              // Exponential backoff
       513 +              time.Sleep(time.Duration(attempt) * time.Second)
       514 +          }
       515 +      }
       516 +      
       517 +      return "", fmt.Errorf("extraction failed after 3 attempts: %w", lastErr)
       518 +  }
