     /// Build a PSPF package from manifest
     pub fn build(
         manifest_path: &Path,
         output_path: &Path,
         launcher_path: Option<&Path>,
         private_key_path: Option<&Path>,
     ) -> Result<()> {
         info!("📦 Building PSPF package: {}", output_path.display());
         
         // Use uv binary as launcher if not specified
         let launcher = match launcher_path {
             Some(path) => path.to_path_buf(),
             None => {
                 let uv_path = env::current_exe()?;
                 info!("✨ Using uv itself as launcher: {}", uv_path.display());
                 uv_path
             }
         };
         
         // Configure build options
         let options = BuildOptions {
             launcher_bin: Some(launcher),
             skip_verification: false,
             private_key_path: private_key_path.map(|p| p.to_path_buf()),
             public_key_path: None,  // Will be derived from private key
         };
         
         // Call flavor-rs builder
         builder::build(manifest_path, output_path, options)?;
         
         info!("✅ Package built: {}", output_path.display());
         Ok(())
     }
